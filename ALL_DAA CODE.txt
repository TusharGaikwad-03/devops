1. Fibonacci Series – Recursive and Non-Recursive
Aim: Write a C++ program to calculate Fibonacci numbers using recursive and non-recursive methods and analyse their time and space complexity.

ANS:- 


#include <iostream>
using namespace std;

// Recursive function
int fibR(int n)
{
    if(n <= 1)
        return n;

    return fibR(n-1) + fibR(n-2);
}

// Non-recursive function
int fibNR(int n)
{
    int a = 0, b = 1, c;

    for(int i = 0; i < n; i++)
    {
        cout << a << " ";
        c = a + b;
        a = b;
        b = c;
    }
    return 0;
}

int main()
{
    int n;
    cout << "Enter number: ";
    cin >> n;

    cout << "Recursive Fibonacci: ";
    for(int i = 0; i < n; i++)
        cout << fibR(i) << " ";

    cout << "\nNon-Recursive Fibonacci: ";
    fibNR(n);

    return 0;
}

2. Huffman Encoding Using Greedy Strategy
Aim: Write a C++ program to implement Huffman Encoding using a greedy strategy.

ANS:-

#include <iostream>
#include <queue>
using namespace std;

struct Node
{
    char ch;
    int freq;
    Node *left, *right;

    Node(char c, int f)
    {
        ch = c;
        freq = f;
        left = right = NULL;
    }
};

struct Compare
{
    bool operator()(Node* a, Node* b)
    {
        return a->freq > b->freq;
    }
};

void print(Node* root, string code)
{
    if(!root)
        return;

    if(root->left == NULL && root->right == NULL)
        cout << root->ch << " : " << code << endl;

    print(root->left, code + "0");
    print(root->right, code + "1");
}

int main()
{
    priority_queue<Node*, vector<Node*>, Compare> pq;

    int n, freq;
    char ch;

    cout << "Enter number of characters: ";
    cin >> n;

    for(int i = 0; i < n; i++)
    {
        cin >> ch >> freq;
        pq.push(new Node(ch, freq));
    }

    while(pq.size() > 1)
    {
        Node* l = pq.top();
        pq.pop();

        Node* r = pq.top();
        pq.pop();

        Node* parent = new Node('$', l->freq + r->freq);
        parent->left = l;
        parent->right = r;

        pq.push(parent);
    }

    cout << "Huffman Codes:\n";
    print(pq.top(), "");

    return 0;
}

3. Fractional Knapsack Using Greedy Strategy
Aim: Write a C++ program to solve the Fractional Knapsack problem using a greedy strategy.

Logic: Calculate profit/weight ratio and select items having the highest ratio first. Fractions of items are allowed.

ANS:-


#include <iostream>
#include <algorithm>
using namespace std;

struct Item
{
    int profit, weight;
};

bool compare(Item a, Item b)
{
    return (double)a.profit/a.weight >
           (double)b.profit/b.weight;
}

int main()
{
    int n, capacity;
    Item a[20];

    cout << "Enter number of items: ";
    cin >> n;

    for(int i = 0; i < n; i++)
    {
        cin >> a[i].profit >> a[i].weight;
    }

    cout << "Enter capacity: ";
    cin >> capacity;

    sort(a, a+n, compare);

    double total = 0;

    for(int i = 0; i < n; i++)
    {
        if(capacity >= a[i].weight)
        {
            total += a[i].profit;
            capacity -= a[i].weight;
        }
        else
        {
            total += (double)a[i].profit *
                     capacity / a[i].weight;
            break;
        }
    }

    cout << "Maximum Profit = " << total;

    return 0;
}

4. 0/1 Knapsack Using Dynamic Programming
Aim: Write a C++ program to solve the 0/1 Knapsack problem using Dynamic Programming.

ANS:-

#include <iostream>
using namespace std;

int main()
{
    int n, W;
    int p[20], wt[20];
    int dp[21][101] = {0};

    cout << "Enter number of items: ";
    cin >> n;

    cout << "Enter profits:\n";
    for(int i = 1; i <= n; i++)
        cin >> p[i];

    cout << "Enter weights:\n";
    for(int i = 1; i <= n; i++)
        cin >> wt[i];

    cout << "Enter capacity: ";
    cin >> W;

    for(int i = 1; i <= n; i++)
    {
        for(int w = 0; w <= W; w++)
        {
            if(wt[i] <= w)
                dp[i][w] = max(p[i] + dp[i-1][w-wt[i]],
                               dp[i-1][w]);
            else
                dp[i][w] = dp[i-1][w];
        }
    }

    cout << "Maximum Profit = " << dp[n][W];

    return 0;
}

5. N-Queen Problem Using Backtracking
Aim: Write a C++ program to solve the N-Queen problem using backtracking.

ANS:-


#include <iostream>
using namespace std;

int board[20][20], n;

bool safe(int row, int col)
{
    for(int i = 0; i < row; i++)
        if(board[i][col])
            return false;

    for(int i = row-1, j = col-1; i >= 0 && j >= 0; i--, j--)
        if(board[i][j])
            return false;

    for(int i = row-1, j = col+1; i >= 0 && j < n; i--, j++)
        if(board[i][j])
            return false;

    return true;
}

bool solve(int row)
{
    if(row == n)
        return true;

    for(int col = 0; col < n; col++)
    {
        if(safe(row, col))
        {
            board[row][col] = 1;

            if(solve(row+1))
                return true;

            board[row][col] = 0;
        }
    }

    return false;
}

int main()
{
    cout << "Enter N: ";
    cin >> n;

    if(n < 1 || n > 20)
    {
        cout << "Invalid N";
        return 0;
    }

    if(solve(0))
    {
        cout << "N Queen Matrix:\n";

        for(int i = 0; i < n; i++)
        {
            for(int j = 0; j < n; j++)
                cout << board[i][j] << " ";

            cout << endl;
        }
    }
    else
        cout << "Solution does not exist";

    return 0;
}
